  #score{color:#00ffcc}
  #hp{color:#ff5577}

  #restartBtn:hover{background:#00ccff;transform:scale(1.05)}

<div id="wrap">
  <canvas id="game" width="480" height="640"></canvas>
  <div id="ui"><div id="score">SCORE: 0</div><div id="hp">HP: 100</div></div>
  <div id="overlay" class="hidden">
    <h1>GAME OVER</h1>
    <p id="finalScore">最終分數: 0</p>
    <p style="font-size:15px;color:#8aa">按 R 鍵 或 點擊按鈕重新開始</p>
    <button id="restartBtn">重新開始 (R)</button>
  </div>
  <div id="hint">方向鍵 / WASD 移動 &nbsp;|&nbsp; 空格 發射</div>

<script>
const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
const W = canvas.width, H = canvas.height;
const scoreEl = document.getElementById('score');
const hpEl = document.getElementById('hp');
const overlay = document.getElementById('overlay');
const finalScoreEl = document.getElementById('finalScore');
const restartBtn = document.getElementById('restartBtn');

let keys = {};
let player, bullets, enemies, particles, scoreTexts, stars;
let score, hp, gameOver, enemyTimer, elapsed, shootCooldown;

function initStars(){
  stars = [];
  for(let i=0;i<120;i++){
    stars.push({x:Math.random()*W, y:Math.random()*H, s:Math.random()*2+0.5, v:Math.random()*30+10, tw:Math.random()*Math.PI*2});
  }
}
function reset(){
  player = {x:W/2, y:H-80, w:36, h:28, speed:320};
  bullets = []; enemies = []; particles = []; scoreTexts = [];
  score = 0; hp = 100; gameOver = false;
  enemyTimer = 0; elapsed = 0; shootCooldown = 0;
  overlay.classList.add('hidden');
  updateUI();
}
function updateUI(){
  scoreEl.textContent = 'SCORE: ' + score;
  hpEl.textContent = 'HP: ' + Math.max(0, hp);
}
function spawnEnemy(){
  const speedBase = 90 + elapsed * 4;
  const w = 30 + Math.random()*14;
  enemies.push({
    x: Math.random()*(W-w-20)+10,
    y: -40,
    w: w, h: 26,
    speed: speedBase + Math.random()*60,
    hp: 1,
    wob: Math.random()*Math.PI*2
  });
}
function shoot(){
  bullets.push({x: player.x-6, y: player.y-14, w:4, h:14, speed:520});
  bullets.push({x: player.x+6, y: player.y-14, w:4, h:14, speed:520});
}
function explode(x,y,color,count){
  for(let i=0;i<count;i++){
    const a = Math.random()*Math.PI*2;
    const sp = Math.random()*220+40;
    particles.push({x,y,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,life:0.5+Math.random()*0.4,maxLife:0.9,color,r:Math.random()*3+2});
  }
}
function drawPlayer(){
  const {x,y}=player;
  ctx.save(); ctx.translate(x,y);
  ctx.fillStyle='#00e5ff';
  ctx.beginPath(); ctx.moveTo(0,-18); ctx.lineTo(-16,12); ctx.lineTo(0,6); ctx.lineTo(16,12); ctx.closePath(); ctx.fill();
  ctx.fillStyle='#0088aa'; ctx.fillRect(-20,4,8,10); ctx.fillRect(12,4,8,10);
  ctx.fillStyle='#ffffff'; ctx.beginPath(); ctx.arc(0,-4,5,0,Math.PI*2); ctx.fill();
  ctx.fillStyle='#00e5ff'; ctx.beginPath(); ctx.arc(0,-4,2.5,0,Math.PI*2); ctx.fill();
  ctx.restore();
}
function drawEnemy(e){
  ctx.save(); ctx.translate(e.x+e.w/2, e.y+e.h/2);
  ctx.fillStyle='#ff3344';
  ctx.beginPath(); ctx.moveTo(0,14); ctx.lineTo(-e.w/2,-10); ctx.lineTo(0,-4); ctx.lineTo(e.w/2,-10); ctx.closePath(); ctx.fill();
  ctx.fillStyle='#aa0022'; ctx.fillRect(-e.w/2,-12,6,8); ctx.fillRect(e.w/2-6,-12,6,8);
  ctx.fillStyle='#ffaaaa'; ctx.beginPath(); ctx.arc(0,2,4,0,Math.PI*2); ctx.fill();
  ctx.restore();
}
function rectsOverlap(a,b){
  return a.x < b.x+b.w && a.x+a.w > b.x && a.y < b.y+b.h && a.y+a.h > b.y;
}
function damagePlayer(amount){
  if(gameOver) return;
  hp -= amount;
  explode(player.x, player.y, '#ff5577', 18);
  updateUI();
  if(hp<=0){ hp=0; updateUI(); endGame(); }
}
function endGame(){
  gameOver = true;
  finalScoreEl.textContent = '最終分數: ' + score;
  overlay.classList.remove('hidden');
}
let lastTime = 0;
function loop(t){
  requestAnimationFrame(loop);
  const dt = Math.min((t-lastTime)/1000 || 0, 0.05);
  lastTime = t;
  if(gameOver){ render(dt); return; }
  update(dt);
  render(dt);
}
function update(dt){
  elapsed += dt;
  // stars
  for(let s of stars){ s.y += s.v*dt; s.tw += dt*3; if(s.y>H){s.y=0;s.x=Math.random()*W;} }
  // player move
  let dx=0, dy=0;
  if(keys['ArrowLeft']||keys['a']||keys['A']) dx-=1;
  if(keys['ArrowRight']||keys['d']||keys['D']) dx+=1;
  if(keys['ArrowUp']||keys['w']||keys['W']) dy-=1;
  if(keys['ArrowDown']||keys['s']||keys['S']) dy+=1;
  if(dx!==0&&dy!==0){dx*=0.7071;dy*=0.7071;}
  player.x = Math.max(24, Math.min(W-24, player.x + dx*player.speed*dt));
  player.y = Math.max(H*0.35, Math.min(H-30, player.y + dy*player.speed*dt));
  // shoot
  shootCooldown -= dt;
  if((keys[' ']||keys['Spacebar']) && shootCooldown<=0){ shoot(); shootCooldown=0.18; }
  // spawn enemies, interval shortens over time
  enemyTimer -= dt;
  const interval = Math.max(0.25, 0.9 - elapsed*0.012);
  if(enemyTimer<=0){ spawnEnemy(); enemyTimer = interval; }
  // bullets
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i]; b.y -= b.speed*dt;
    if(b.y<-20) bullets.splice(i,1);
  }
  // enemies
  for(let i=enemies.length-1;i>=0;i--){
    const e=enemies[i];
    e.wob += dt*3;
    e.y += e.speed*dt;
    e.x += Math.sin(e.wob)*30*dt;
    // reached bottom
    if(e.y > H-10){
      enemies.splice(i,1);
      damagePlayer(10);
      continue;
    }
    // hit player
    const pr = {x:player.x-16,y:player.y-16,w:32,h:28};
    const er = {x:e.x,y:e.y,w:e.w,h:e.h};
    if(rectsOverlap(pr,er)){
      enemies.splice(i,1);
      damagePlayer(20);
      continue;
    }
  }
  // bullet vs enemy
  for(let i=bullets.length-1;i>=0;i--){
    const b=bullets[i];
    const br={x:b.x-b.w/2,y:b.y,w:b.w,h:b.h};
    let hit=false;
    for(let j=enemies.length-1;j>=0;j--){
      const e=enemies[j];
      const er={x:e.x,y:e.y,w:e.w,h:e.h};
      if(rectsOverlap(br,er)){
        explode(e.x+e.w/2, e.y+e.h/2, '#ffaa00', 16);
        scoreTexts.push({x:e.x+e.w/2, y:e.y, text:'+100', life:0.8});
        score+=100; updateUI();
        enemies.splice(j,1);
        hit=true; break;
      }
    }
    if(hit) bullets.splice(i,1);
  }
  // particles
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i]; p.life-=dt;
    if(p.life<=0){particles.splice(i,1);continue;}
    p.x+=p.vx*dt; p.y+=p.vy*dt; p.vx*=0.98; p.vy*=0.98;
  }
  for(let i=scoreTexts.length-1;i>=0;i--){
    const s=scoreTexts[i]; s.life-=dt; s.y-=40*dt;
    if(s.life<=0) scoreTexts.splice(i,1);
  }
}
function render(dt){
  ctx.fillStyle='#050814'; ctx.fillRect(0,0,W,H);
  // stars
  for(let s of stars){
    const a = 0.4 + 0.6*Math.abs(Math.sin(s.tw));
    ctx.globalAlpha=a; ctx.fillStyle='#ffffff';
    ctx.fillRect(s.x,s.y,s.s,s.s);
  }
  ctx.globalAlpha=1;
  if(!gameOver) drawPlayer();
  // bullets
  ctx.fillStyle='#ccff00';
  for(let b of bullets){
    ctx.fillRect(b.x-b.w/2,b.y,b.w,b.h);
    ctx.fillStyle='#ffffff'; ctx.fillRect(b.x-1,b.y,2,6); ctx.fillStyle='#ccff00';
  }
  for(let e of enemies) drawEnemy(e);
  for(let p of particles){
    ctx.globalAlpha=Math.max(0,p.life/p.maxLife);
    ctx.fillStyle=p.color; ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.fill();
  }
  ctx.globalAlpha=1;
  ctx.fillStyle='#00ffcc'; ctx.font='bold 16px Arial'; ctx.textAlign='center';
  for(let s of scoreTexts){
    ctx.globalAlpha=Math.max(0,s.life/0.8);
    ctx.fillText(s.text,s.x,s.y);
  }
  ctx.globalAlpha=1;
}
window.addEventListener('keydown', e=>{
  if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight',' '].includes(e.key)) e.preventDefault();
  keys[e.key]=true;
  if((e.key==='r'||e.key==='R'||e.key==='R'.toLowerCase()) && gameOver) reset();
});
window.addEventListener('keyup', e=>{ keys[e.key]=false; });
restartBtn.addEventListener('click', reset);
// touch support
let touchX=null, touchY=null, touching=false;
canvas.addEventListener('touchstart', e=>{ touching=true; const t=e.touches[0]; const r=canvas.getBoundingClientRect(); touchX=t.clientX-r.left; touchY=t.clientY-r.top; keys[' ']=true; },{passive:true});
canvas.addEventListener('touchmove', e=>{
  if(!touching) return;
  const t=e.touches[0]; const r=canvas.getBoundingClientRect();
  const nx=(t.clientX-r.left)*(W/r.width); const ny=(t.clientY-r.top)*(H/r.height);
  player.x += (nx-player.x)*0.3; player.y += (ny-player.y)*0.3;
  e.preventDefault();
},{passive:false});
canvas.addEventListener('touchend', ()=>{ touching=false; keys[' ']=false; });

initStars();
reset();
requestAnimationFrame(loop);
</script>
